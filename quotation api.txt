case "Select_Cust":
    {
        List<object> customerList = new List<object>();
        string selectQuery = "SELECT [cust_name],[phone_no],[mail_id],[gst_no],[pan_no],[address],[district],[pin_code],[state] FROM [Data_Master].[dbo].[Customer_Master] WHERE state = 'Tamil Nadu'";
        sql_cmd = new SqlCommand(selectQuery, sql_con);
        sql_read = sql_cmd.ExecuteReader();
        if (sql_read.HasRows)
        {
            while (sql_read.Read())
            {
                customerList.Add(new
                {
                    cust_name = sql_read.GetValue(0).ToString().Trim(),
                    phone_no = sql_read.GetValue(1).ToString().Trim(),
                    mail_id = sql_read.GetValue(2).ToString().Trim(),
                    gst_no = sql_read.GetValue(3).ToString().Trim(),
                    pan_no = sql_read.GetValue(4).ToString().Trim(),
                    address = sql_read.GetValue(5).ToString().Trim(),
                    district = sql_read.GetValue(6).ToString().Trim(),
                    pin_code = sql_read.GetValue(7).ToString().Trim(),
                    state = sql_read.GetValue(8).ToString().Trim(),
                });
            }
        }
        sql_con.Close();
        jsonout.json_status = "2";
        jsonout.error_msg = "Success";
        jsonout.data = customerList;
        break;
    }
